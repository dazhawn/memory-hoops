import React, { useRef, useMemo } from 'react';
import { useFrame } from '@react-three/fiber';
import { MeshReflectorMaterial, Stars } from '@react-three/drei';
import * as THREE from 'three';

type GymType = 'classic' | 'outdoor' | 'arcade';

interface CourtProps {
  gym: GymType;
  netRippling?: boolean;
}

interface HoopProps {
  position?: [number, number, number];
  netRippling?: boolean;
}

// Shared Hoop Component
function Hoop({ position = [0, 4, -8], netRippling = false }: HoopProps) {
  const netGroupRef = useRef<THREE.Group>(null!);
  const rippleRef = useRef(0);

  const netStrands = useMemo(() => {
    const strands: React.ReactNode[] = [];
    const count = 16;
    const rimRadius = 0.45;
    const netHeight = 1.2;
    const bottomRadius = 0.18;

    for (let i = 0; i < count; i++) {
      const angle = (i / count) * Math.PI * 2;
      const x = Math.cos(angle) * rimRadius;
      const z = Math.sin(angle) * rimRadius;
      const bottomX = Math.cos(angle) * bottomRadius;
      const bottomZ = Math.sin(angle) * bottomRadius;

      strands.push(
        <mesh
          key={i}
          position={[x, -netHeight / 2, z]}
          rotation={[0, angle, 0]}
        >
          <cylinderGeometry args={[0.008, 0.008, netHeight, 8]} />
          <meshStandardMaterial
            color="#ff8800"
            metalness={0.3}
            roughness={0.7}
          />
        </mesh>
      );
    }
    return strands;
  }, []);

  useFrame((_, delta) => {
    if (!netGroupRef.current) return;

    if (netRippling) {
      rippleRef.current = Math.min(rippleRef.current + delta * 8, 1);
    } else {
      rippleRef.current = Math.max(rippleRef.current - delta * 3, 0);
    }

    const sway = Math.sin(Date.now() * 0.008) * 0.015 * rippleRef.current;
    netGroupRef.current.rotation.y = sway * 0.6;
    netGroupRef.current.children.forEach((child, i) => {
      if (child instanceof THREE.Mesh) {
        child.rotation.z = Math.sin(Date.now() * 0.01 + i) * 0.08 * rippleRef.current;
      }
    });
  });

  return (
    <group position={position}>
      {/* Vertical Pole */}
      <mesh position={[0, 2, 0]}>
        <cylinderGeometry args={[0.12, 0.15, 4.2, 32]} />
        <meshPhysicalMaterial
          color="#aaaaaa"
          metalness={0.95}
          roughness={0.15}
          clearcoat={0.3}
        />
      </mesh>

      {/* Horizontal Arm */}
      <mesh position={[0, 4.1, -0.8]} rotation={[0, 0, Math.PI * 0.5]}>
        <cylinderGeometry args={[0.08, 0.08, 1.8, 32]} />
        <meshPhysicalMaterial
          color="#aaaaaa"
          metalness={0.95}
          roughness={0.15}
          clearcoat={0.3}
        />
      </mesh>

      {/* Backboard Mounting Bracket */}
      <mesh position={[0, 4.3, -1.6]}>
        <boxGeometry args={[0.6, 0.15, 0.3]} />
        <meshStandardMaterial color="#555555" metalness={0.9} roughness={0.2} />
      </mesh>

      {/* Backboard - Frosted Acrylic Glass */}
      <mesh position={[0, 4.1, -2.4]}>
        <boxGeometry args={[2.4, 1.8, 0.08]} />
        <meshPhysicalMaterial
          color="#e0f0ff"
          metalness={0.1}
          roughness={0}
          transmission={0.85}
          thickness={0.8}
          clearcoat={1}
          clearcoatRoughness={0.1}
          envMapIntensity={0.6}
        />
      </mesh>

      {/* Backboard Border / Frame */}
      <mesh position={[0, 4.1, -2.45]}>
        <boxGeometry args={[2.55, 1.92, 0.06]} />
        <meshStandardMaterial color="#222222" metalness={0.8} roughness={0.3} />
      </mesh>

      {/* Rim */}
      <group position={[0, 3.3, -2.0]}>
        <mesh rotation={[Math.PI * 0.5, 0, 0]}>
          <torusGeometry args={[0.45, 0.04, 24, 64]} />
          <meshPhysicalMaterial
            color="#ff5500"
            metalness={0.9}
            roughness={0.2}
            clearcoat={0.8}
            clearcoatRoughness={0.1}
          />
        </mesh>
        {/* Rim net hooks (subtle) */}
        <mesh position={[0, 0.05, 0]} rotation={[Math.PI * 0.5, 0, 0]}>
          <torusGeometry args={[0.47, 0.015, 12, 64]} />
          <meshStandardMaterial color="#333333" metalness={0.95} roughness={0.4} />
        </mesh>
      </group>

      {/* Net */}
      <group ref={netGroupRef} position={[0, 3.3, -2.0]}>
        {netStrands}
      </group>
    </group>
  );
}

// Main Court Component
export default function Court({ gym, netRippling = false }: CourtProps) {
  const floorRef = useRef<THREE.Mesh>(null!);

  return (
    <>
      {/* Floor */}
      <group>
        {/* Main Court Floor */}
        <mesh ref={floorRef} rotation={[-Math.PI * 0.5, 0, 0]} position={[0, 0, 0]} receiveShadow>
          <planeGeometry args={[28, 50]} />
          <MeshReflectorMaterial
            resolution={1024}
            blur={[400, 300]}
            mixBlur={8}
            mixStrength={
              gym === 'arcade' ? 2.8 : gym === 'outdoor' ? 1.6 : 1.2
            }
            roughness={
              gym === 'arcade' ? 0.04 : gym === 'outdoor' ? 0.6 : 0.25
            }
            metalness={gym === 'arcade' ? 0.1 : 0}
            color={
              gym === 'classic'
                ? '#d4a373'
                : gym === 'outdoor'
                ? '#1a1a1a'
                : '#0a0a0a'
            }
            reflectorOffset={0.02}
          />
        </mesh>

        {/* Court Lines (emissive) */}
        <mesh rotation={[-Math.PI * 0.5, 0, 0]} position={[0, 0.02, 0]} receiveShadow>
          <planeGeometry args={[26, 48]} />
          <meshStandardMaterial
            color="#ffffff"
            emissive="#ffffff"

            opacity={0.9}

          </mesh>

          {/* Bleachers */}
          {Array.from({ length: 5 }).map((_, i) => (

            </mesh>
          ))}
        </>
      )}

      {/* Outdoor Night Court */}
      {gym === 'outdoor' && (

          <mesh position={[-18, 22, -35]}>

          <Stars radius={300} depth={50} count={800} factor={4} saturation={0} fade />

          {/* Floodlight Poles */}
          {[-8, 8].map((x, i) => (

              {/* Pole */}

              </mesh>
              {/* Glow Lens */}
              <mesh position={[0, 13.4, 0.7]}>
                <planeGeometry args={[1.4, 0.6]} />
                <meshBasicMaterial color="#aaffff" />
              </mesh>
              <pointLight
                color="#aaffff"
                intensity={8}
                distance={45}
                decay={1.5}
                position={[0, 13.2, 1]}
                castShadow
              />
            </group>
          ))}

          {/* Chain-link Fence (simplified) */}

            <group key={x} position={[x, 4, -5]}>

      {/* Arcade Cyber Court */}
      {gym === 'arcade' && (

          {/* Neon Floor Grid */}
          <mesh rotation={[-Math.PI * 0.5, 0, 0]} position={[0, 0.03, 0]}>
            <planeGeometry args={[28, 50]} />

              color="#00ffff"
              emissive="#00ffff"
              emissiveIntensity={3}

          {/* Neon Wall Strips */}
          {[

          ].map((wall, i) => (

              {/* Bottom neon */}
              <mesh position={[wall.pos[0], 1.2, wall.pos[2]]} rotation={wall.rot}>

                <planeGeometry args={[wall.pos[0] === 0 ? 28 : 50, 0.4]} />

          {/* Dark Walls */}
          {[
            [-14, 5, 0],

          ].map((pos, i) => (
